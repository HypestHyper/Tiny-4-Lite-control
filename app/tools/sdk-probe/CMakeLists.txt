# Builds the SDK probe against a local copy of OBSBOT's SDK (not part of this repository).
#   cmake -S . -B build -DOBSBOT_SDK=$HOME/obsbot-sdk/libdev_v2.1.0_8 && cmake --build build
cmake_minimum_required(VERSION 3.10)
project(obsbot_sdk_probe CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT OBSBOT_SDK)
  message(FATAL_ERROR "Set -DOBSBOT_SDK=/path/to/libdev_v2.1.0_8 (the unzipped SDK)")
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  set(SDK_LIB "${OBSBOT_SDK}/linux/arm64-release")
else()
  set(SDK_LIB "${OBSBOT_SDK}/linux/x86_64-release")
endif()
add_executable(sdk-probe probe.cpp)
target_include_directories(sdk-probe PRIVATE "${OBSBOT_SDK}/include")
target_link_directories(sdk-probe PRIVATE "${SDK_LIB}")
target_link_libraries(sdk-probe PRIVATE dev pthread ${CMAKE_DL_LIBS})
set_target_properties(sdk-probe PROPERTIES BUILD_RPATH "${SDK_LIB}")
